let currentQ = 0;

let answers = "";

 

// 切换页面

function goPage(id){

  document.querySelectorAll('.page').forEach(p=>p.classList.remove('active'));

  document.getElementById('page'+id).classList.add('active');

  if(id===2) renderQuestion();

}

 

//渲染题目

function renderQuestion(){

  const item = questions[currentQ];

  document.getElementById('qIndex').innerText = currentQ+1;

  const box = document.getElementById('questionBox');

  let html = `<div class="text-[18px] font-medium text-[#3b5c3f] mb-6">${item.q}</div>`;

  item.opts.forEach((opt,i)=>{

    html += `<div class="option" onclick="selectOpt('${opt[0]}')">${opt}</div>`;

  })

  box.innerHTML = html;

}

 

function selectOpt(val){

  answers += val;

  currentQ++;

  if(currentQ >= questions.length){

    showResult();

  }else{

    renderQuestion();

  }

}

 

function showResult(){

  goPage(3);

  const res = resultMap[answers];

  document.getElementById('resTitle').innerText = res.title;

  document.getElementById('resStar').innerText = res.star;

  document.getElementById('resDesc').innerText = res.desc;

}

 

//下载海报

function downloadPoster(){

  const posterDom = document.getElementById('poster');

  html2canvas(posterDom,{scale:2,backgroundColor:null}).then(canvas=>{

    const a = document.createElement('a');

    a.href = canvas.toDataURL('image/png');

    a.download = "抹茶冰椰奶_风味报告.png";

    a.click();

  })

}

 

//重置
